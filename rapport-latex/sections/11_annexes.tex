\section{Annexe A - Commandes utiles}

\subsection*{Backend}
\begin{lstlisting}[style=codeblock,language=bash]
cd football-academy
mvn clean test
mvn clean package -DskipTests
mvn spring-boot:run
\end{lstlisting}

\subsection*{Frontend}
\begin{lstlisting}[style=codeblock,language=bash]
cd moez_project
flutter pub get
flutter analyze
flutter test
flutter run
\end{lstlisting}

\subsection*{Chatbot}
\begin{lstlisting}[style=codeblock,language=bash]
cd chatbot/chatbot
python -m venv env_py10
pip install -r requirements.txt
python manage.py migrate
python manage.py runserver 127.0.0.1:8000
\end{lstlisting}
